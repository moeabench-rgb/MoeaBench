    def constraints(self,f,parameter = 1,f_c=[]):
        f_constraints=np.array(f)
        f_c = np.array([np.sum([ f_c**2  for  f_c in f_constraints[linha,0:f_constraints.shape[1]]])-parameter for index,linha in enumerate(range(f_constraints.shape[0]))  ])
        return f_c


    def eval_cons(self,f):
        M_constraints = self.constraints(f)
        eval = M_constraints == 0
        return f[eval]


    def get_Points(self):
        return np.array([*np.random.random((self.get_P(), self.get_N()))*1.0])


    def F1(self,M,th,Gxm):
       theta = list(map(lambda TH: np.cos(TH), th[0:(M-1)]))
       return (1+Gxm)*np.prod(np.column_stack(theta ), axis = 1).reshape(Gxm.shape[0],1)


    def F2(self,M,th,Gxm):
        theta = list(map(lambda TH: np.cos(TH), th[0:(M-2)]))
        return (1+Gxm)*np.prod(np.column_stack(theta ), axis = 1).reshape(Gxm.shape[0],1)*np.column_stack(np.sin(th[(M-2):(M-1)]))


    def F3(self,M,th,Gxm):
        theta = list(map(lambda TH: np.cos(TH), th[0:(M-3)]))
        return (1+Gxm)*np.prod(np.column_stack(theta ), axis = 1).reshape(Gxm.shape[0],1)*np.column_stack(np.sin(th[(M-3):(M-2)]))


    def Fm(self,M,th,Gxm):
        return (1+Gxm)*np.column_stack(np.sin(th[0:1]))


    def get_method(self,enum):
        return self.llist_E_DTLZ[enum]


    def param_F(self):
        dict_F = {
                    self.get_method(0) : self.F1,
                    self.get_method(1) : self.F2,
                    self.get_method(2) : self.F3,
                    self.get_method(3) : self.Fm
                  }
        return dict_F


    def calc_F_M(self,Fi,M):
        if Fi == 1:
            return self.get_method(0)
        elif Fi == 2 and M > 2:
            return self.get_method(1)
        elif Fi >= 3 and Fi <= M-1 and M > 3:
            return self.get_method(2)
        elif Fi == M:
            return self.get_method(3)


    def calc_TH(self,X,Gxm,M):
        return [X[:,Xi:Xi+1]*np.pi/2 if Xi == 0 else (np.pi/(4*(1+Gxm))*(1+2*Gxm*X[:,Xi:Xi+1]))  for Xi in range(0,M-1)]


    def calc_f(self,X,G):
        vet_F_M = [self.calc_F_M(F,self.get_M()) for F, i in enumerate(range(0,self.get_M()), start = 1)]
        return np.column_stack(list(map(lambda Key: self.param_F()[Key](self.get_M(),self.calc_TH(X,G,self.get_M()),G),vet_F_M)))


    def calc_g(self,X):
        return np.sum((X[:,self.get_M()-1:]-0.5)**2, axis = 1).reshape(X.shape[0],1)

    🧩MoeaBench - implement the BaseBenchmark method

    The evaluation(x,n_ieq) method is a mandatory implementation of the BaseBenchmark class.

    experiment.problem

    This variable should contain the object of the benchmark problem class that implements the Evaluation(x,n_ieq) method.

    Evaluation method

    Evaluation(x,n_ieq) method must receive the decision variables of the evolutionary algorithm across generations. For each array received, it must calculate the decision variables imposed by the problem implementation. The user can also add a method within evaluation to assign constraints or penalties to these objectives. The evolutionary algorithm receives this array of solutions from the evaluation method and uses these results to evolve its population across generations. This cycle repeats until the evolutionary algorithm completes all generations.

    🧩MoeaBench - notes

    The BaseBenchmark class also provides native methods: get_M() (integer with the number of objectives), get_N() (integer with the number of decision variables), get_P() (integer with number of samples for problem).
